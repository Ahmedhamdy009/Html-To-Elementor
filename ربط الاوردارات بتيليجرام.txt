const ORDERS_SHEET_NAME = 'Orders';
const ORDER_TRACKING_META_KEY = '_order_tracking_number';

/* =========================================================
   1) FULL ORDERS SYNC (المزامنة الشاملة وحذف الممسوح)
   ========================================================= */

function syncOrdersMaster() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(ORDERS_SHEET_NAME);

  if (!sheet) throw new Error('Sheet "Orders" not found.');

  const orders = fetchAllOrders_();
  Logger.log('Fetched ' + orders.length + ' orders from WooCommerce.');

  reconcileOrdersSheet_(sheet, orders);
  Logger.log('Orders sync completed successfully.');
}

function fetchAllOrders_() {
  const all = [];
  let page = 1;

  while (true) {
    const batch = wooRequest_('GET', '/orders?status=any&per_page=50&page=' + page);
    if (!batch || !Array.isArray(batch) || !batch.length) break;
    batch.forEach(function(order) { all.push(order); });
    if (batch.length < 50) break;
    page++;
  }
  return all;
}

function reconcileOrdersSheet_(sheet, orders) {
  const lastRow = sheet.getLastRow();
  const wooIdsMap = {};
  orders.forEach(function(o) { wooIdsMap[String(o.id)] = o; });

  // 1. حذف الأوردرات من الشيت إذا تم مسحها من ووكومرس
  if (lastRow >= 2) {
    const existingIds = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = existingIds.length - 1; i >= 0; i--) {
      const rowId = String(existingIds[i][0] || '').trim();
      if (rowId && !wooIdsMap[rowId]) {
        sheet.deleteRow(i + 2);
        Logger.log('Deleted removed order #' + rowId + ' from Sheet.');
      }
    }
  }

  // 2. تحديث القائم وإضافة الجديد
  upsertOrdersToSheet_(sheet, orders);
}

function upsertOrdersToSheet_(sheet, orders) {
  const lastRow = sheet.getLastRow();
  const existingIds = {};

  if (lastRow >= 2) {
    const idValues = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    idValues.forEach(function(row, index) {
      const id = String(row[0] || '').trim();
      if (id) existingIds[id] = index + 2;
    });
  }

  const rowsToAppend = [];
  const newOrdersToNotify = [];

  orders.forEach(function(order) {
    const rowValues = buildOrderRow_(order);
    const orderId = String(order.id);

    if (existingIds[orderId]) {
      const rowNumber = existingIds[orderId];
      const currentTracking = String(sheet.getRange(rowNumber, 13).getValue() || '').trim();
      if (!rowValues[12] && currentTracking) {
        rowValues[12] = currentTracking;
      }
      sheet.getRange(rowNumber, 1, 1, rowValues.length).setValues([rowValues]);
    } else {
      rowsToAppend.push(rowValues);
      newOrdersToNotify.push(order);
    }
  });

  if (rowsToAppend.length) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend);

    // إرسال إشعار تليجرام تلقائياً للأوردر الجديد
    newOrdersToNotify.forEach(function(newOrder) {
      try {
        sendOrderNotificationToTelegram(newOrder);
      } catch (e) {
        Logger.log('Telegram notify error for order ' + newOrder.id + ': ' + e.message);
      }
    });
  }
}

/* =========================================================
   2) ROW BUILDER & HELPERS
   ========================================================= */
function buildOrderRow_(order) {
  const billing = order.billing || {};
  const customerName = String(((billing.first_name || '') + ' ' + (billing.last_name || '')).trim());
  const phone = String(billing.phone || '').trim();
  const email = String(billing.email || '').trim();
  const address = formatOrderAddress_(billing);

  const lineItems = order.line_items || [];
  const productsText = lineItems
    .map(function(item) {
      const meta = (item.meta_data || []).filter(function(m) { return !m.key.startsWith('_'); }).map(function(m) { return m.value; }).join(' / ');
      return item.name + (meta ? ' (' + meta + ')' : '') + ' × ' + item.quantity;
    })
    .join(' | ');

  const quantity = lineItems.reduce(function(sum, item) { return sum + (Number(item.quantity) || 0); }, 0);
  const total = Number(order.total) || 0;
  const paymentMethod = String(order.payment_method_title || '').trim();
  const status = String(order.status || '').trim();
  const customerNote = String(order.customer_note || '').trim();
  const trackingNumber = getOrderTrackingFromMeta_(order);
  const orderDate = formatOrderDate_(order.date_created);

  return [
    order.id, orderDate, customerName, phone, email, address,
    productsText, quantity, total, paymentMethod, status, customerNote, trackingNumber
  ];
}

function formatOrderAddress_(billing) {
  const parts = [billing.address_1, billing.address_2, billing.city, billing.state, billing.postcode, billing.country]
    .filter(function(p) { return p && String(p).trim(); });
  return parts.join(', ');
}

function formatOrderDate_(dateString) {
  if (!dateString) return '';
  try {
    const date = new Date(dateString);
    const timezone = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
    return Utilities.formatDate(date, timezone, 'yyyy-MM-dd HH:mm');
  } catch (error) { return dateString; }
}

function getOrderTrackingFromMeta_(order) {
  const meta = order.meta_data || [];
  for (const item of meta) {
    if (item.key === ORDER_TRACKING_META_KEY) return String(item.value || '').trim();
  }
  return '';
}

/* =========================================================
   3) EDIT IN SHEET → PUSH TO WOOCOMMERCE
   ========================================================= */

function updateOrdersMaster(e) {
  if (!e || !e.range) return;
  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== ORDERS_SHEET_NAME || range.getRow() < 2) return;
  const column = range.getColumn();
  if (column !== 11 && column !== 13) return;

  const row = range.getRow();
  const orderId = sheet.getRange(row, 1).getValue();
  if (!orderId) return;

  try {
    if (column === 11) {
      const newStatus = String(range.getValue() || '').trim().toLowerCase();
      if (!newStatus) return;
      wooRequest_('PUT', '/orders/' + orderId, { status: newStatus });
      Logger.log('✅ تم تحديث حالة الطلب #' + orderId + ' في ووكومرس إلى: ' + newStatus);
    }
    if (column === 13) {
      const trackingNumber = String(range.getValue() || '').trim();
      wooRequest_('PUT', '/orders/' + orderId, {
        meta_data: [{ key: ORDER_TRACKING_META_KEY, value: trackingNumber }]
      });
      Logger.log('✅ تم تحديث رقم تتبع الطلب #' + orderId + ' في ووكومرس.');
    }
  } catch (error) {
    Logger.log('❌ خطأ أثناء التحديث: ' + error.message);
  }
}

/* =========================================================
   4) TELEGRAM BOT & NOTIFICATIONS WITH SMART WHATSAPP
   ========================================================= */

function getTelegramBotToken_() {
  const token = PropertiesService.getScriptProperties().getProperty('TELEGRAM_BOT_TOKEN');
  if (!token) throw new Error('TELEGRAM_BOT_TOKEN missing in Script Properties.');
  return token;
}

function getTelegramChatId_() {
  const chatId = PropertiesService.getScriptProperties().getProperty('TELEGRAM_CHAT_ID');
  if (!chatId) throw new Error('TELEGRAM_CHAT_ID missing.');
  return chatId;
}

function telegramApiRequest_(method, payload) {
  const token = getTelegramBotToken_();
  const url = 'https://api.telegram.org/bot' + token + '/' + method;
  const options = {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(payload || {}),
    muteHttpExceptions: true
  };
  const response = UrlFetchApp.fetch(url, options);
  return JSON.parse(response.getContentText());
}

function processTelegramUpdates() {
  const props = PropertiesService.getScriptProperties();
  const lastOffset = Number(props.getProperty('TELEGRAM_LAST_OFFSET') || 0);

  const json = telegramApiRequest_('getUpdates', {
    offset: lastOffset + 1,
    limit: 50,
    timeout: 0
  });

  if (!json.ok || !json.result || !json.result.length) return;

  let maxOffset = lastOffset;

  json.result.forEach(function(update) {
    if (update.update_id > maxOffset) maxOffset = update.update_id;
    if (update.callback_query) {
      handleTelegramCallbackDirect_(update.callback_query);
    }
  });

  props.setProperty('TELEGRAM_LAST_OFFSET', String(maxOffset));
}

// دالة تجهيز رسالة الواتساب الذكية بروابط التأكيد والإلغاء المباشرة للعميل
function buildCustomerWhatsAppUrl_(order, type) {
  const billing = order.billing || {};
  let rawPhone = String(billing.phone || '').replace(/[^0-9]/g, '');
  if (!rawPhone) return '';

  // ضبط كود الدولة لأرقام مصر تلقائياً
  if (rawPhone.startsWith('01')) {
    rawPhone = '20' + rawPhone.substring(1);
  } else if (rawPhone.startsWith('1') && rawPhone.length === 10) {
    rawPhone = '20' + rawPhone;
  } else if (!rawPhone.startsWith('20') && rawPhone.length === 11) {
    rawPhone = '20' + rawPhone.substring(1);
  }

  const customerName = ((billing.first_name || '') + ' ' + (billing.last_name || '')).trim() || 'عميلنا العزيز';
  const total = Number(order.total || 0) + ' ' + (order.currency_symbol || order.currency || 'ج.م');

  // سحب رابط موقعك الحقيقي تلقائياً من الإعدادات
  const props = PropertiesService.getScriptProperties();
  const siteUrl = (props.getProperty('SITE_URL') || '').replace(/\/$/, '');
  const orderKey = order.order_key || '';
  const confirmUrl = siteUrl + '/?wc_order_action=confirm&order_id=' + order.id + '&key=' + orderKey;
  const cancelUrl = siteUrl + '/?wc_order_action=cancel&order_id=' + order.id + '&key=' + orderKey;

  let message = '';

  if (type === 'confirm') {
    message = 'أهلاً يا ' + customerName + '\n' +
              'تم تأكيد طلبك رقم #' + order.id + ' بنجاح!\n' +
              'جاري تجهيز طلبك للشحن في أسرع وقت. شكراً لتسوقك معنا 🛍️';
  } else if (type === 'reject') {
    message = 'أهلاً يا ' + customerName + '\n' +
              'نعتذر منك، تم إلغاء طلبك رقم #' + order.id + '.\n' +
              'إذا كان لديك أي استفسار يسعدنا تواصلك معنا دائماً 🌸';
  } else {
    // رسالة التأكيد والإلغاء التفاعلية بنقرة واحدة
    message = 'أهلاً يا ' + customerName + ' 👋\n' +
              'تم استلام طلبك رقم #' + order.id + ' بإجمالي (' + total + ').\n\n' +
              '🟢 لتأكيد شحن طلبك بنقرة واحدة:\n' +
              confirmUrl + '\n\n' +
              '🔴 أو لإلغاء الطلب:\n' +
              cancelUrl + '\n\n' +
              'شكراً لتسوقك معنا 🌸';
  }

  return 'https://wa.me/' + rawPhone + '?text=' + encodeURIComponent(message);
}

function handleTelegramCallbackDirect_(callbackQuery) {
  const callbackId = callbackQuery.id;
  const data = String(callbackQuery.data || '');
  const message = callbackQuery.message;
  const chatId = message ? message.chat.id : getTelegramChatId_();
  const messageId = message ? message.message_id : null;

  const isConfirm = data.startsWith('confirm_');
  const isReject = data.startsWith('reject_');
  if (!isConfirm && !isReject) return;

  const orderId = data.replace('confirm_', '').replace('reject_', '');

  try {
    let order = wooRequest_('GET', '/orders/' + orderId);
    if (!order || !order.id) return;

    if (isConfirm) {
      wooRequest_('PUT', '/orders/' + orderId, { status: 'processing' });
      updateSheetOrderStatusDirectly_(orderId, 'processing');

      const waConfirmUrl = buildCustomerWhatsAppUrl_(order, 'confirm');
      const keyboard = waConfirmUrl ? {
        inline_keyboard: [[{ text: '💬 إرسال رسالة التأكيد للعميل واتساب', url: waConfirmUrl }]]
      } : { inline_keyboard: [] };

      if (messageId) {
        order.status = 'processing';
        const updatedText = formatOrderTelegramMessage_(order) + '\n\n━━━━━━━━━━━━━━━\n🟢 <b>تم تأكيد الطلب بنجاح</b> (Processing)';
        telegramApiRequest_('editMessageText', {
          chat_id: chatId,
          message_id: messageId,
          text: updatedText,
          parse_mode: 'HTML',
          reply_markup: keyboard
        });
      }

      telegramApiRequest_('answerCallbackQuery', {
        callback_query_id: callbackId,
        text: '✅ تم تأكيد الطلب #' + orderId + ' بنجاح!'
      });

    } else if (isReject) {
      wooRequest_('PUT', '/orders/' + orderId, { status: 'cancelled' });
      updateSheetOrderStatusDirectly_(orderId, 'cancelled');

      const waRejectUrl = buildCustomerWhatsAppUrl_(order, 'reject');
      const keyboard = waRejectUrl ? {
        inline_keyboard: [[{ text: '💬 إرسال إشعار الإلغاء للعميل واتساب', url: waRejectUrl }]]
      } : { inline_keyboard: [] };

      if (messageId) {
        order.status = 'cancelled';
        const updatedText = formatOrderTelegramMessage_(order) + '\n\n━━━━━━━━━━━━━━━\n🔴 <b>تم إلغاء الطلب</b> (Cancelled)';
        telegramApiRequest_('editMessageText', {
          chat_id: chatId,
          message_id: messageId,
          text: updatedText,
          parse_mode: 'HTML',
          reply_markup: keyboard
        });
      }

      telegramApiRequest_('answerCallbackQuery', {
        callback_query_id: callbackId,
        text: '🔴 تم إلغاء الطلب #' + orderId
      });
    }
  } catch (e) {
    Logger.log('Callback error: ' + e.message);
  }
}

function sendOrderNotificationToTelegram(order) {
  const chatId = getTelegramChatId_();
  const messageText = formatOrderTelegramMessage_(order);
  const waUrl = buildCustomerWhatsAppUrl_(order, 'check');

  const keyboardRows = [
    [
      { text: '🟢 تأكيد الطلب (Processing)', callback_data: 'confirm_' + order.id },
      { text: '🔴 إلغاء الطلب (Cancel)', callback_data: 'reject_' + order.id }
    ]
  ];

  if (waUrl) {
    keyboardRows.push([{ text: '💬 إرسال رسالة التأكيد والإلغاء الذكية للعميل واتساب', url: waUrl }]);
  }

  return telegramApiRequest_('sendMessage', {
    chat_id: chatId,
    text: messageText,
    parse_mode: 'HTML',
    reply_markup: { inline_keyboard: keyboardRows }
  });
}

function escapeHtml_(text) {
  if (!text) return '';
  return String(text)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;');
}

function formatOrderTelegramMessage_(order) {
  const billing = order.billing || {};
  const customerName = escapeHtml_(((billing.first_name || '') + ' ' + (billing.last_name || '')).trim() || 'عميل');
  const phone = escapeHtml_(billing.phone || 'غير متوفر');
  const address = escapeHtml_(formatOrderAddress_(billing) || 'غير متوفر');
  const currency = escapeHtml_(order.currency_symbol || order.currency || 'EGP');

  const lineItems = (order.line_items || []).map(function(item) {
    const meta = (item.meta_data || []).filter(function(m) { return !m.key.startsWith('_'); }).map(function(m) { return m.value; }).join(' / ');
    return '• ' + escapeHtml_(item.name) + (meta ? ' (' + escapeHtml_(meta) + ')' : '') + ' × ' + item.quantity + ' — ' + item.total + ' ' + currency;
  }).join('\n');

  const subtotal = (order.line_items || []).reduce(function(s, i) { return s + Number(i.total || 0); }, 0);
  const shipping = Number(order.shipping_total || 0);
  const total = Number(order.total || 0);
  const paymentMethod = escapeHtml_(order.payment_method_title || 'غير محدد');
  const orderDate = formatOrderDate_(order.date_created);

  return '🛒 <b>طلب جديد #' + order.id + '</b>\n\n' +
         '👤 <b>العميل:</b> ' + customerName + '\n' +
         '📞 <b>الهاتف:</b> <code>' + phone + '</code>\n' +
         '📍 <b>العنوان:</b> ' + address + '\n\n' +
         '📦 <b>المنتجات:</b>\n' + lineItems + '\n\n' +
         '💰 <b>المجموع الفرعي:</b> ' + subtotal + ' ' + currency + '\n' +
         '🚚 <b>الشحن:</b> ' + shipping + ' ' + currency + '\n' +
         '💵 <b>الإجمالي:</b> <b>' + total + ' ' + currency + '</b>\n' +
         '💳 <b>طريقة الدفع:</b> ' + paymentMethod + '\n' +
         '🕐 <b>التاريخ:</b> ' + orderDate;
}

function updateSheetOrderStatusDirectly_(orderId, newStatus) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(ORDERS_SHEET_NAME);
    if (!sheet) return;
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) return;

    const idValues = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < idValues.length; i++) {
      if (String(idValues[i][0]).trim() === String(orderId).trim()) {
        sheet.getRange(i + 2, 11).setValue(newStatus);
        break;
      }
    }
  } catch (e) {
    Logger.log('Direct status update error: ' + e.message);
  }
}
